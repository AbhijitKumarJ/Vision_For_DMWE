```typescript
import { EventEmitter } from 'events';

// ============================================================================
// 1. SCHEMAS & TYPES (From Volume XI)
// ============================================================================

enum PayloadType { DESCRIPTOR, CAPABILITY_ADVERT, INTERACTION_OBJ, NODE_OFFLINE }
enum Verb { SELECT = "SELECT", MOVE = "MOVE", ROTATE = "ROTATE" }
enum Target { CURRENT_OBJECT = "CURRENT_OBJECT", IOT_DEVICE = "IOT_DEVICE" }

interface MeshEnvelope {
    messageId: string;
    senderId: string;
    payloadType: PayloadType;
    payload?: any;
}

interface CapabilityAdvertisement {
    capabilityId: string;
    version: string;
    qoc: { latencyMs: number; accuracyScore: number };
}

interface InteractionObject {
    interactionId: string;
    intent: { verb: Verb; target: Target };
    origin: { capabilityId: string; providerId: string };
    confidence: number;
}

// ============================================================================
// 2. THE TRANSPORT LAYER (Volume VIII)
// ============================================================================

class MeshRadio extends EventEmitter {
    broadcast(packet: MeshEnvelope) {
        setTimeout(() => this.emit('rx_packet', packet), 5); // 5ms latency
    }
}
const localRadio = new MeshRadio();

// ============================================================================
// 3. PHYSICAL MODULES (Volume III)
// ============================================================================

class SmartRingNode {
    public nodeId = "DMWE-RNG-9942";
    constructor(private radio: MeshRadio) {}

    public boot() {
        console.log(`\n💍 [RING] Powering on. Running secure boot...`);
        this.radio.broadcast({
            messageId: `msg-rng-1`, senderId: this.nodeId, payloadType: PayloadType.DESCRIPTOR,
            payload: { deviceClass: "SENSOR_NODE" }
        });

        this.radio.broadcast({
            messageId: `msg-rng-2`, senderId: this.nodeId, payloadType: PayloadType.CAPABILITY_ADVERT,
            payload: { capabilityId: "interaction.pointer.precision", version: "1.0", qoc: { latencyMs: 8, accuracyScore: 0.98 } }
        });
    }

    public simulateAirPinch() {
        console.log(`\n👆 [RING] User performed 'Air Pinch' gesture.`);
        this.radio.broadcast({
            messageId: `msg-rng-3`, senderId: this.nodeId, payloadType: PayloadType.INTERACTION_OBJ,
            payload: {
                interactionId: `ix-rng-${Date.now()}`,
                intent: { verb: Verb.SELECT, target: Target.IOT_DEVICE },
                origin: { capabilityId: "interaction.pointer.precision", providerId: this.nodeId },
                confidence: 0.95
            }
        });
    }

    public die() {
        console.log(`\n🪫 [RING] BATTERY CRITICAL. Shutting down abruptly!`);
        this.radio.broadcast({ messageId: `msg-rng-4`, senderId: this.nodeId, payloadType: PayloadType.NODE_OFFLINE });
    }
}

class SmartGlassesNode {
    public nodeId = "DMWE-GLS-1123";
    constructor(private radio: MeshRadio) {}

    public boot() {
        console.log(`\n👓 [GLASSES] Powering on. Running secure boot...`);
        this.radio.broadcast({
            messageId: `msg-gls-1`, senderId: this.nodeId, payloadType: PayloadType.DESCRIPTOR,
            payload: { deviceClass: "VISION_NODE" }
        });

        this.radio.broadcast({
            messageId: `msg-gls-2`, senderId: this.nodeId, payloadType: PayloadType.CAPABILITY_ADVERT,
            payload: { capabilityId: "interaction.gaze.select", version: "2.0", qoc: { latencyMs: 45, accuracyScore: 0.85 } }
        });
    }

    public simulateProlongedBlink() {
        console.log(`\n👁️  [GLASSES] User performed 'Prolonged Blink' gesture.`);
        this.radio.broadcast({
            messageId: `msg-gls-3`, senderId: this.nodeId, payloadType: PayloadType.INTERACTION_OBJ,
            payload: {
                interactionId: `ix-gls-${Date.now()}`,
                intent: { verb: Verb.SELECT, target: Target.IOT_DEVICE },
                origin: { capabilityId: "interaction.gaze.select", providerId: this.nodeId },
                confidence: 0.88
            }
        });
    }
}

// ============================================================================
// 4. MESH OS COORDINATOR (Volumes IV & V)
// ============================================================================

class MeshOSPrimary {
    private capabilityRegistry: Array<{ providerId: string, capabilityId: string }> = [];
    private appSubscriptions: Array<{ verb: Verb, callback: Function }> = [];
    private isFused = false;

    constructor(private radio: MeshRadio) {
        this.radio.on('rx_packet', (packet: MeshEnvelope) => this.handlePacket(packet));
        console.log(`🧠 [MESH OS] Coordinator online. Listening to radio...`);
    }

    private handlePacket(packet: MeshEnvelope) {
        switch (packet.payloadType) {
            case PayloadType.CAPABILITY_ADVERT:
                this.capabilityRegistry.push({ providerId: packet.senderId, capabilityId: packet.payload.capabilityId });
                console.log(`⚙️  [MESH OS] Registered Capability: '${packet.payload.capabilityId}' from ${packet.senderId}`);
                this.evaluateCapabilityFusion(); // Check if we can fuse capabilities
                break;

            case PayloadType.INTERACTION_OBJ:
                console.log(`⚡ [MESH OS] Intent Engine received Interaction: [${packet.payload.intent.verb}] from ${packet.senderId}`);
                this.routeToApps(packet.payload);
                break;

            case PayloadType.NODE_OFFLINE:
                console.log(`⚠️  [MESH OS] Health Manager: Node ${packet.senderId} dropped from mesh!`);
                // Remove dead node's capabilities from registry
                this.capabilityRegistry = this.capabilityRegistry.filter(c => c.providerId !== packet.senderId);
                this.evaluateCapabilityFusion(); // Re-evaluate fusion state
                break;
        }
    }

    // Volume V, Chapter 12: Capability Fusion
    private evaluateCapabilityFusion() {
        const hasRing = this.capabilityRegistry.some(c => c.capabilityId === "interaction.pointer.precision");
        const hasGlasses = this.capabilityRegistry.some(c => c.capabilityId === "interaction.gaze.select");

        if (hasRing && hasGlasses && !this.isFused) {
            console.log(`🌀 [MESH OS] CAPABILITY FUSION: Ring + Glasses combined into 'interaction.spatial_pointer.fused'`);
            this.isFused = true;
        } else if ((!hasRing || !hasGlasses) && this.isFused) {
            console.log(`📉 [MESH OS] FALLBACK ACTIVATED: Fusion broken. Falling back to individual capabilities.`);
            this.isFused = false;
        }
    }

    public subscribeToIntent(verb: Verb, callback: Function) {
        this.appSubscriptions.push({ verb, callback });
    }

    private routeToApps(interaction: InteractionObject) {
        // In a real MeshOS, this would filter by confidence and active fusion states.
        this.appSubscriptions.forEach(sub => {
            if (sub.verb === interaction.intent.verb) sub.callback(interaction);
        });
    }
}

// ============================================================================
// 5. THE DEVELOPER APPLICATION (Volume IX - Intent-Oriented Programming)
// ============================================================================

function startSmartHomeApp(meshOS: MeshOSPrimary) {
    console.log(`\n📱 [APP] Smart Home App launching...`);
    
    // The App subscribes to the INTENT. It does not know or care about Rings or Glasses.
    meshOS.subscribeToIntent(Verb.SELECT, (interaction: InteractionObject) => {
        if (interaction.intent.target === Target.IOT_DEVICE) {
            console.log(`💡 [APP TRIGGERED] Turning ON the Smart Lamp!`);
            console.log(`   (Triggered by capability: ${interaction.origin.capabilityId} with ${interaction.confidence * 100}% confidence)`);
        }
    });
}

// ============================================================================
// 6. RUN THE SIMULATION SCENARIO
// ============================================================================

async function runSimulation() {
    console.log("==========================================");
    console.log("   DMWE MeshOS Expanded Scenario          ");
    console.log("==========================================\n");

    const meshOS = new MeshOSPrimary(localRadio);
    startSmartHomeApp(meshOS);

    const myGlasses = new SmartGlassesNode(localRadio);
    const myRing = new SmartRingNode(localRadio);

    // 1. Boot both devices
    myGlasses.boot();
    setTimeout(() => myRing.boot(), 20);

    // 2. User uses the Ring to turn on the lamp (Optimal State / Fused)
    setTimeout(() => myRing.simulateAirPinch(), 100);

    // 3. DISASTER STRIKES: Ring battery dies
    setTimeout(() => myRing.die(), 200);

    // 4. User still needs to turn off the lamp, instinctively blinks at it (Fallback)
    setTimeout(() => myGlasses.simulateProlongedBlink(), 300);
}

runSimulation();
```

### Key Architectural Concepts Proven Here:

1. **Volume V, Chapter 12: Capability Fusion:** When the Ring joins, the `MeshOS` detects that both a high-precision IMU (Ring) and Gaze Tracking (Glasses) are present. It dynamically creates a composite `interaction.spatial_pointer.fused` capability.
2. **Volume IV, Chapter 16: Fault Tolerance:** When the ring unexpectedly dies (`myRing.die()`), the MeshOS Health Manager instantly quarantines the node and deregisters its capabilities. It realizes the Fusion is broken and gracefully degrades back to individual capabilities. 
3. **Volume IX, Chapter 1: Intent-Oriented Programming (IOP):** This is the most crucial part. Look at the `startSmartHomeApp` function. **The application code never changes.** It doesn't crash when the ring dies, and it doesn't require an `if (glasses.areAvailable())` block. It simply asks the MeshOS to tell it when a user expresses the intent to `SELECT` an `IOT_DEVICE`, and the platform handles the rest.
